\chapter{Sequences}
\section{Convergence}

\dfn[convergence]{Convergence}{
A sequence $(a_n)_{n\in\bbN}$ in $\bbR$ converges to $L\in\bbR$ if,
for every $\veps>0$, there exists $N\in\bbN$ such that
\[
  n\geq N \quad\Longrightarrow\quad |a_n-L|<\veps.
\]
}

\ex[reciprocal]{A convergent sequence}{
The sequence $a_n=1/n$ converges to zero. Given $\veps>0$, choose
$N>1/\veps$; then $n\geq N$ implies $|1/n|\leq 1/N<\veps$.
}

\thm[unique-limit]{Uniqueness of limits}{
A convergent real sequence has exactly one limit.
}
\begin{proof}
Suppose $a_n\to L$ and $a_n\to M$ with $L\neq M$.
Set $\veps=|L-M|/3$. For sufficiently large $n$,
\[
  |L-M|\leq |L-a_n|+|a_n-M|
  <2\veps=\frac23|L-M|,
\]
a contradiction. Thus $L=M$.
\end{proof}

\rmk[limit-notation]{Notation}{
Theorem~\ref{th:unique-limit} justifies writing ``the limit'' of a
convergent sequence. Definition~\ref{def:convergence} gives its meaning.
}

\section{Boundedness}
\lem[bounded]{Convergent sequences are bounded}{
Every convergent real sequence is bounded.
}
\begin{proof}
Suppose $a_n\to L$. Choose $N$ so that $n\geq N$ implies $|a_n-L|<1$,
and set
\[
  B=\max\{|a_1|,\ldots,|a_N|,|L|+1\}.
\]
For $n<N$, the definition of $B$ gives $|a_n|\leq B$.
For $n\geq N$, the triangle inequality gives
$|a_n|\leq |a_n-L|+|L|<1+|L|\leq B$.
\end{proof}

\cor[unbounded]{A divergence criterion}{
An unbounded real sequence cannot converge in $\bbR$.
}
